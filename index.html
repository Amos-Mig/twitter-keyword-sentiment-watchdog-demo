What the real invocation looks like

# watchdog.py — excerpt (~30 of 380 lines). The full file ships with the kit.
WATCH = load_watchlist("watch.toml")     # keywords, sources, quiet hours
LEX   = load_lexicon()                   # built-in VADER-style scorer

for post in stream(WATCH.sources, WATCH.keywords):
    score = LEX.score(post.text)         # -1.0 … +1.0
    history.save(post, score)
    rolling.push(score)
    if spike_detector(rolling, threshold=2.6):
        alert(
            title = f"{WATCH.name}: {'negative' if score < 0 else 'positive'} spike",
            body  = post.permalink,
            via   = WATCH.webhooks,      # slack / email / generic
        )
digest.maybe_send(history, now)          # quiet-day digest at 18:00
